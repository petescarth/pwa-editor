export function getEditorTheme(isDark: boolean): Extension {
  return isDark
    ? [darkEditorTheme, syntaxHighlighting(darkHighlightStyle)]
    : [lightEditorTheme, syntaxHighlighting(lightHighlightStyle)];
}
